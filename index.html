Assignment 3: Grade Calculator

let score = Number(prompt("Enter your test score:"));
let grade;

if (score >= 90) {
    grade = "A";
} else if (score >= 80) {
    grade = "B";
} else if (score >= 70) {
    grade = "C";
} else if (score >= 60) {
    grade = "D";
} else {
    grade = "F";
}

switch (grade) {
    case "A": grade = "A"; break;
    case "B": grade = "B"; break;
    case "C": grade = "C"; break;
    case "D": grade = "D"; break;
    default: grade = "F";
}

let status = score >= 50 ? "Passed" : "Failed";

console.log("Score:", score);
console.log("Grade:", grade);
console.log("Status:", status);

Assignment 6: Array Methods

const products = [
    { name: "Laptop", price: 800 },
    { name: "Mouse", price: 25 },
    { name: "Keyboard", price: 45 }
];

console.log("-- All Products (forEach) --");

products.forEach(product => {
    console.log(
        `Item: ${product.name} - Price: $${product.price}`
    );
});

console.log("-- Filtered Products (Price < $50) --");

const filteredProducts =
    products.filter(product => product.price < 50);

console.log(filteredProducts);

console.log("-- Mapped Product Tags (map) --");

const mappedProducts = products.map(
    product =>
        `${product.name.toUpperCase()} costs $${product.price}`
);

console.log(mappedProducts);

Project: Inventory Management System

const inventory = [
    { name: "Laptop", price: 800 },
    { name: "Mouse", price: 25 }
];

let productName = prompt("Enter product name:");
let productPrice = Number(prompt("Enter product price:"));

if (
    productName &&
    productName.trim() !== "" &&
    Number.isFinite(productPrice) &&
    productPrice > 0
) {
    inventory.push({
        name: productName.trim(),
        price: productPrice
    });
} else {
    console.log("Invalid input! Product was not added.");
}

function getCategory(price) {
    if (price < 100) return "Budget";
    else if (price >= 500) return "Premium";
    else return "Standard";
}

const calculateTotal = (items) => {
    let total = 0;

    for (let i = 0; i < items.length; i++) {
        if (!items[i]) continue;
        total += items[i].price;
    }

    return total;
};

function calculateFinalPrice(price, tax = 0, discount = 0) {
    let finalPrice = price + (price * tax / 100);
    finalPrice -= finalPrice * discount / 100;
    return finalPrice;
}
